  <script>
    // --- GENEL SAYFA YÖNETİMİ ---
    function goToPage(pageNum) {
      document.querySelectorAll('.page').forEach(page => page.classList.remove('active'));
      document.getElementById('page' + pageNum).classList.add('active');

      if (pageNum === 6) {
        startGame();
      } else {
        stopGame();
      }
    }

    function openBox(pageNum) {
      goToPage(pageNum);
    }

    // --- SAYFA 1: ŞİFRE MANTIĞI ---
    let enteredPin = "";
    const correctPin = "0807";

    function pressKey(key) {
      if (key === 'X') {
        enteredPin = enteredPin.slice(0, -1);
      } else if (key === 'OK') {
        checkPin();
        return;
      } else {
        if (enteredPin.length < 4) {
          enteredPin += key;
        }
      }

      updateHearts();

      if (enteredPin.length === 4) {
        setTimeout(checkPin, 200);
      }
    }

    function updateHearts() {
      for (let i = 1; i <= 4; i++) {
        const dot = document.getElementById('dot' + i);
        if (i <= enteredPin.length) {
          dot.classList.add('filled');
        } else {
          dot.classList.remove('filled');
        }
      }
    }

    function checkPin() {
      if (enteredPin === correctPin) {
        goToPage(2);
        enteredPin = "";
        updateHearts();
      } else {
        const page1 = document.getElementById('page1');
        page1.classList.add('shake');
        setTimeout(() => page1.classList.remove('shake'), 400);
        enteredPin = "";
        updateHearts();
      }
    }

    // --- SAYFA 5: MÜZİK MANTIĞI ---
    const music = document.getElementById('bgMusic');
    const vinyl = document.getElementById('vinylBtn');

    function toggleMusic() {
      if (music.paused) {
        music.play();
        vinyl.classList.add('spinning');
      } else {
        music.pause();
        vinyl.classList.remove('spinning');
      }
    }

    function stopMusic() {
      music.pause();
      music.currentTime = 0;
      vinyl.classList.remove('spinning');
    }

    // --- SAYFA 6: OYUN MANTIĞI ---
    let score = 0;
    let heartInterval = null;
    let activeHeart = null;

    function startGame() {
      score = 0;
      document.getElementById('scoreBoard').style.display = 'block';
      document.getElementById('scoreBoard').innerText = `Yakalanan Kalp: ${score} / 5`;
      document.getElementById('winText').style.display = 'none';
      
      clearHearts();
      spawnHeart();
    }

    function clearHearts() {
      if (activeHeart) activeHeart.remove();
      if (heartInterval) clearInterval(heartInterval);
    }

    function stopGame() {
      clearHearts();
    }

    function spawnHeart() {
      const gameArea = document.getElementById('gameArea');
      if (score >= 5) return;

      if (activeHeart) activeHeart.remove();

      const heart = document.createElement('div');
      heart.className = 'floating-heart';
      heart.innerText = '💖';

      // Rastgele Konumlama (Kutu içinde kalacak şekilde)
      const maxX = gameArea.clientWidth - 50;
      const maxY = gameArea.clientHeight - 60;
      
      const randomX = Math.max(10, Math.floor(Math.random() * maxX));
      const randomY = Math.max(40, Math.floor(Math.random() * maxY));

      heart.style.left = randomX + 'px';
      heart.style.top = randomY + 'px';

      // Yakalama Tıklaması
      heart.onclick = function() {
        score++;
        document.getElementById('scoreBoard').innerText = `Yakalanan Kalp: ${score} / 5`;
        heart.remove();

        if (score >= 5) {
          document.getElementById('scoreBoard').style.display = 'none';
          document.getElementById('winText').style.display = 'block';
          if (heartInterval) clearInterval(heartInterval);
        } else {
          spawnHeart();
        }
      };

      gameArea.appendChild(activeHeart = heart);

      // Kalbin kaçma hızı (Sürekli yer değiştirir)
      if (heartInterval) clearInterval(heartInterval);
      heartInterval = setInterval(() => {
        if (score < 5) spawnHeart();
      }, 700); // 700ms - Hızlı ve yakalaması tatlı zorlukta
    }
  </script>
